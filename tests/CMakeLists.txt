cmake_minimum_required(VERSION 3.16)
project(controller-tests LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
enable_testing()
add_executable(gamepad-identity gamepad_identity.cpp)
target_compile_features(gamepad-identity PRIVATE cxx_std_11)
target_include_directories(gamepad-identity PRIVATE ../app ../moonlight-common-c/moonlight-common-c/src)
target_link_libraries(gamepad-identity PRIVATE PkgConfig::SDL2)
add_test(NAME gamepad-identity COMMAND gamepad-identity)

foreach(protocol_version IN ITEMS old new)
    add_executable(gamepad-capability-${protocol_version} gamepad_capability_compat.cpp)
    target_compile_features(gamepad-capability-${protocol_version} PRIVATE cxx_std_11)
    target_include_directories(gamepad-capability-${protocol_version} PRIVATE ../app ../moonlight-common-c/moonlight-common-c/src)
    if(protocol_version STREQUAL "new")
        target_compile_definitions(gamepad-capability-${protocol_version} PRIVATE TEST_PROTOCOL_HAS_EDGE)
    endif()
    add_test(NAME gamepad-capability-${protocol_version} COMMAND gamepad-capability-${protocol_version})
endforeach()
